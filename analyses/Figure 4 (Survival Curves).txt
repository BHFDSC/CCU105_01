1 = No CKD
2 = CKD
3 = Death
*/

1-2 (CKD onset)
1-3 (death without CKD)
2-3 (death after CKD)
*/

*Create a numeric person id
gen pid = _n

*Use msset to declare and restructure the data for multi-state modelling
msset, id(pid) states(flag_ckd death_flag) times(time_to_ckd survtime)

*msset also returns a default transition matrix - this needs to be stored for later use
mat tmat = r(transmatrix)
mat list tmat

*fit a model by transition
stset _stop, enter(_start) failure(_status) scale(365.25)

*only analyse the transitions to death
keep if _to==3 // 1-3 or 2-3 transitions

/* 
_from=1: no CKD
_from=2: CKD
Effect = HR comparing death hazard when exposed vs. unexposed
*/

*Generate splines for age
rcsgen mi_index_age, df(3) gen(agercs) orthog
global ageknots `r(knots)'
matrix R_age=r(R)

range temptime 0 8 97


*Adjusted model - covid period 1
stpm3 i._from @rcs(mi_index_age, df(3)) i.sex b5.ethnicity i.imd_2019_quintiles i.diabetes_flag i.hypertension_flag i.obesity_flag i.smoking_status2 i.invasivestrategy i.discharge_diagnosis i.aspirin i.betablocker i.statin i.ace i.p2y12 i.aldosterone if covid_period==1, scale(logcumhazard) df(4) tvc(@rcs(mi_index_age, df(3))) dftvc(2) eform
estimates store surv_trans_covid1
